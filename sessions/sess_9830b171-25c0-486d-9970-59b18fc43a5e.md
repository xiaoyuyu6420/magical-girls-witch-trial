- 没 VPS → 迁到 Neon/Supabase 的免费 Postgres 再上 Vercel（Prisma 迁移成本低）

## 二、SEO 怎么给主域供血

先纠正一个预期：单个小站传给主域的权重是慢变量、收益有限。真正有效的是这两招：

- **域名形态选子目录而不是子域**：`你的主域/witch-trial` 比子域 `witch.你的主域` 更能把收录和权重直接记在主域头上（Google 把子域当相对独立的站）。且 portfolio 和作品天然绑定。
- **内链要自然，别做成互链农场**：footer 一行 "built by 你 · 查看我的其他作品” 指回主站就够了。多个自己的站互刷链接会被降权，得不偿失。

**最大的 SEO 机会点**：现在是单页应用，测完的结果不产生 URL。如果给 **16 种人格各做一个静态落地页**（`/result/homura` 这种），每页就是一个长尾关键词着陆页（“晓美焰 型人格”、“which magical girl are you”）——这类 quiz 站的长尾流量是最持久有效的渠道，而且和结果页“去查看全部人格”的跳转完全自然。四语言 × 16 页 = 64 个被索引页面。这是本策略里工作量最大的一项（约一两天），也是回报最大的。

配套小件：Search Console 提交 sitemap、hreflang 四语标签、结构化数据。

## 三、当作品集能做什么

这个项目在作品集里已经具备稀缺性——不是又一个 todo app，而是**有完整工程深度的趣味产品**：服务端防作弊向量计算、12 维匹配算法的调参故事、四语 i18n、E2E 红蓝对抗审计。具体产出四样东西：

1. **Portfolio case study 一页**（核心）：讲三个最吸睛的技术决策——向量撞车怎么调参、为什么算分放服务端、审计抓出的 bug 怎么修。面试官看的是思考过程，不是截图。
2. **GitHub README 补一条 live demo 链接**就齐活了。
3. **一段录屏 GIF / 截图**，一份素材两用：简历里用 + 小红书笔记里用。
4. **一篇技术文章**发掘金/知乎/dev.to：《我用向量匹配做了一个魔法少女人格测试》——文章本身又是一个搜索入口，还能引 GitHub star。

## 四、不挂广告的隐性变现（按性价比排）

1. **结果卡裂变**（已具备）：html-to-image 生成的结果图角落带低调优雅的域名水印 = 免费的传播闭环。建议检查一下水印现在是否存在、是否够低调。
2. **导流主业**：footer 的作者链接把quiz 流量变成 portfolio 访问量——“隐性变现”的正解在这里：变现物是你的能力和服务，不是站点本身。
3. **itch.io 免费上架 web 版**：被列表站和 AI 助手引用是新流量口（2026 年很多人问 ChatGPT“推荐个人格测试”，被引用的前提是被清晰收录）。
4. Ko-fi/爱发电 一行赞助位——可选，收入几乎为零但零氛围破坏。
5. ❌ 不做：affiliate、付费解锁隐藏人格——都会毁掉你要的艺术氛围。

**发布渠道**（气质匹配度排序）：中文圈小红书笔记（测试类内容天然适配，配结果图）、B站动态、V2EX 分享创造节点；英文圈 Reddit 相关社区 + X anime 圈。（这块本想搜一下最新渠道格局，搜索配额刚好耗尽、8 月末重置，以上基于我掌握的行业共识——大方向即“社区+长尾>发布会式 launch"，风险很低。）

37	  for (let i = 0; i < chars.length; i += 3) {
38	    groups.push(chars.slice(i, i + 3).join(""));
39	  }
40	  return groups.join("-");
41	}

51	/** 加权曼哈顿距离 */
52	export function weightedManhattan(a: number[], b: number[], pack: QuizPack = getActivePack()): number {
53	  const codes = dimCodes(pack);
54	  const weights = dimWeights(pack);
55	  if (a.length !== codes.length || b.length !== codes.length) {
56	    throw new Error(`Invalid vector length: a=${a.length}, b=${b.length}, dims=${codes.length}`);
57	  }
58	  let dist = 0;
59	  for (let i = 0; i < codes.length; i++) {
60	    dist += weights[i] * Math.abs(a[i] - b[i]);
61	  }
62	  return dist;
63	}

120	function applyGateBonus(
121	  dimScores: Record<string, number>,
122	  gateValue: string | undefined,
123	  pack: QuizPack,
124	): Record<string, number> {
125	  const scores = { ...dimScores };
126	  if (!gateValue) return scores;
127	  const bonus = pack.rules.gateBonus[gateValue];
128	  if (!bonus) return scores;
129	  for (const [dim, add] of Object.entries(bonus)) {
130	    scores[dim] = Math.min((scores[dim] ?? 0) + add, 6);
131	  }
132	  return scores;
133	}

136	  const scores = dimScores;
137	  return dimCodes(pack).map((code) => {

139	    return scoreToTier(total, pack);
140	  });

143	function resolveSpecialCode(
144	  trigger: string,
145	  gateValue: string,
146	  specialTypes: { code: string }[],
147	  pack: QuizPack,
148	): string | undefined {
149	  // 1) Explicit pack map: SPECIAL_A.destroy → YUKI
150	  const mapped = pack.rules.specialTriggers[trigger]?.[gateValue]
151	    ?? pack.rules.specialTriggers[trigger.toUpperCase()]?.[gateValue];
152	  if (mapped && specialTypes.some((t) => t.code === mapped)) return mapped;
153	
154	  // 2) Direct personality code
155	  if (specialTypes.some((t) => t.code === trigger)) return trigger;
156	
157	  // 3) Legacy SPECIAL_A / SPECIAL_1 index into candidate list (stable sort by code)
158	  const match = /^SPECIAL_([A-Z]|\d+)$/i.exec(trigger);
159	  if (!match) return undefined;
160	
161	  const token = match[1].toUpperCase();
162	  const index = /^\d+$/.test(token)
163	    ? Number(token) - 1
164	    : token.charCodeAt(0) - "A".charCodeAt(0);
165	
166	  const sorted = [...specialTypes].sort((a, b) => a.code.localeCompare(b.code));
167	  // Historical gate filtering: endure → odd indices in insertion order was used;
168	  // keep pack map as primary; legacy fallback uses full sorted list.
169	  return index >= 0 ? sorted[index]?.code : undefined;
170	}

172	export function match(
173	  input: MatchInput,
174	  types: PersonalityTypeInput[],
175	  pack: QuizPack = getActivePack(),
176	): MatchResult {
177	  const allTypes = types.map((t) => ({
178	    ...t,
179	    subtitle: t.subtitle ?? undefined,
180	    keywords: t.keywords ?? undefined,
181	  }));
182	
183	  const regularTypes = allTypes.filter((t) => !t.special && t.group !== "fallback");
184	  const specialTypes = allTypes.filter((t) => t.special);
185	  const unsetType = allTypes.find((t) => t.group === "fallback");
186	  if (!unsetType) {
187	    throw new Error("Missing fallback personality type in database");
188	  }
189	
190	  const dimLen = pack.dimensions.length;
191	
192	  // ① 特殊触发
193	  if (input.triggerFired && input.gateValue) {
194	    const specialCode = resolveSpecialCode(
195	      input.triggerFired,
196	      input.gateValue,
197	      specialTypes,
198	      pack,
199	    );
200	    if (specialCode) {
201	      const t = allTypes.find((p) => p.code === specialCode);
202	      if (!t) {
203	        throw new Error(`Missing special personality type ${specialCode} in database`);
204	      }
205	      const scored = applyGateBonus(input.dimScores, input.gateValue, pack);
206	      const specialUserVec = formatVector(buildUserValues(scored, pack));
207	
208	      return {
209	        code: t.code,
210	        name: t.name,
211	        subtitle: t.subtitle,
212	        slogan: t.slogan,
213	        desc: t.desc,
214	        keywords: t.keywords,
215	        similarity: 100,
216	        userVector: specialUserVec,
217	        templateVector: t.vector,
218	        top3: [{ code: t.code, name: t.name, similarity: 100, translations: t.translations }],
219	        group: "special",
220	        borderType: false,
221	        special: true,
222	        translations: t.translations,
223	        ipCode: t.ipCode ?? undefined,
224	      };
225	    }
226	  }
227	
228	  // ② 计算用户向量
229	  const scored = applyGateBonus(input.dimScores, input.gateValue, pack);
230	  const userValues = buildUserValues(scored, pack);
231	  const userVec = formatVector(userValues);
232	
233	  // ③ 向量匹配
234	  const ranked = regularTypes
235	    .map((t) => {
236	      const tplValues = parseVector(t.vector, dimLen);
237	      const dist = weightedManhattan(userValues, tplValues, pack);
238	      return { type: t, dist, sim: similarity(dist, pack) };
239	    })
240	    .sort((a, b) => a.dist - b.dist);
241	
242	  const top3 = ranked.slice(0, 3);
243	  const best = top3[0];
244	  if (!best) {
245	    return {
246	      code: unsetType.code,
247	      name: unsetType.name,
248	      subtitle: unsetType.subtitle,
249	      slogan: unsetType.slogan,
250	      desc: unsetType.desc,
251	      keywords: unsetType.keywords,
252	      similarity: 0,
253	      userVector: userVec,
254	      templateVector: unsetType.vector,
255	      top3: [],
256	      group: "fallback",
257	      borderType: true,
258	      special: false,
259	      translations: unsetType.translations,
260	      ipCode: unsetType.ipCode ?? undefined,
261	    };
262	  }
263	
264	  // ④ 边界检查
265	  const delta = pack.algo.delta;
266	  const threshold = pack.algo.threshold;
267	  let borderType = false;
268	  const resultCode = best.type.code;
269	
270	  if (top3.length >= 2) {
271	    const gap = best.sim - top3[1].sim;
272	    if (gap < delta) {
273	      if (best.sim < threshold) {
274	        return {
275	          code: unsetType.code,
276	          name: unsetType.name,
277	          subtitle: unsetType.subtitle,
278	          slogan: unsetType.slogan,
279	          desc: unsetType.desc,
280	          keywords: unsetType.keywords,
281	          similarity: best.sim,
282	          userVector: userVec,
283	          templateVector: unsetType.vector,
284	          top3: top3.map((r) => ({
285	            code: r.type.code,
286	            name: r.type.name,
287	            similarity: r.sim,
288	            translations: r.type.translations,
289	          })),
290	          group: "fallback",
291	          borderType: true,
292	          special: false,
293	          translations: unsetType.translations,
294	          ipCode: unsetType.ipCode ?? undefined,
295	        };
296	      }
297	      borderType = true;
298	    }
299	  }
300	
301	  return {
302	    code: resultCode,
303	    name: best.type.name,
304	    subtitle: best.type.subtitle,
305	    slogan: best.type.slogan,
306	    desc: best.type.desc,
307	    keywords: best.type.keywords,
308	    similarity: best.sim,
309	    userVector: userVec,
310	    templateVector: best.type.vector,
311	    top3: top3.map((r) => ({
312	      code: r.type.code,
313	      name: r.type.name,
314	      similarity: r.sim,
315	      translations: r.type.translations,
316	    })),
317	    group: best.type.group,
318	    borderType,
319	    special: false,
320	    translations: best.type.translations,
321	    ipCode: best.type.ipCode ?? undefined,
322	  };
323	}
